(function () { 'use strict'; function applyTheme(theme) { var effective = theme || 'dark'; document.documentElement.setAttribute('data-theme', effective); document.body.setAttribute('data-theme', effective); try { localStorage.setItem('theme', effective); } catch (e) {} var meta = document.querySelector('meta[name="theme-color"]'); if (meta) meta.setAttribute('content', effective === 'light' ? '#eef0f6' : '#07070c'); } function detectPreferredTheme() { try { var stored = localStorage.getItem('theme'); if (stored === 'light' || stored === 'dark') return stored; } catch (e) {} if (window.matchMedia && window.matchMedia('(prefers-color-scheme: light)').matches) return 'light'; return 'dark'; } function wireTheme() { var btn = document.getElementById('theme-toggle'); if (!btn || btn.dataset.bpWired) return; btn.dataset.bpWired = '1'; applyTheme(detectPreferredTheme()); btn.addEventListener('click', function () { var cur = (document.documentElement.getAttribute('data-theme') || 'dark'); applyTheme(cur === 'light' ? 'dark' : 'light'); }); } function lazyLoadImages() { var images = document.querySelectorAll('img:not([loading])'); for (var i = 0; i < images.length; i++) { var img = images[i]; if (!img.dataset.lazyApplied) { img.setAttribute('loading', 'lazy'); img.setAttribute('decoding', 'async'); img.dataset.lazyApplied = '1'; } } } function wireKeyboardNavigation() { var navItems = function () { return Array.prototype.slice.call(document.querySelectorAll('.bp-nav__link, .bp-drawer__link, .bp-drawer__cta')); }; document.addEventListener('keydown', function (e) { if (!/^Arrow(Right|Left|Down|Up)$/.test(e.key)) return; var items = navItems(); if (!items.length) return; var start = items.indexOf(document.activeElement); if (start === -1) return; var step = (e.key === 'ArrowRight' || e.key === 'ArrowDown') ? 1 : -1; var next = (start + step + items.length) % items.length; items[next].focus(); e.preventDefault(); }, true); } function ensureMobileNavVisibility() { var nav = document.querySelector('.bp-nav'); var toggle = document.querySelector('.bp-nav-toggle'); if (!nav || !toggle) return; if (window.innerWidth <= 767) { nav.style.display = 'none'; toggle.style.display = 'grid'; } } function boot() { wireTheme(); lazyLoadImages(); wireKeyboardNavigation(); ensureMobileNavVisibility(); window.addEventListener('resize', ensureMobileNavVisibility, { passive: true }); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', boot, { once: true }); } else { boot(); } })();
